import time
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader

# Set seed for reproducibility
torch.manual_seed(42)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")

# ==========================================
# 1. FIXED HYPERPARAMETERS & DATA PIPELINE
# ==========================================
BATCH_SIZE = 128
EPOCHS = 15
LEARNING_RATE = 0.001
NUM_CLASSES = 100  # Updated from 10 to 100

# Channel-wise mean and standard deviation calculated for CIFAR-100
CIFAR100_MEAN = (0.5071, 0.4867, 0.4408)
CIFAR100_STD = (0.2675, 0.2565, 0.2761)

transform_train = transforms.Compose([
    transforms.RandomCrop(32, padding=4),
    transforms.RandomHorizontalFlip(),
    transforms.ToTensor(),
    transforms.Normalize(CIFAR100_MEAN, CIFAR100_STD),
])

transform_test = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize(CIFAR100_MEAN, CIFAR100_STD),
])

# Download and load CIFAR-100 dataset
trainset = torchvision.datasets.CIFAR100(root='./data', train=True, download=True, transform=transform_train)
train_loader = DataLoader(trainset, batch_size=BATCH_SIZE, shuffle=True, num_workers=2)

testset = torchvision.datasets.CIFAR100(root='./data', train=False, download=True, transform=transform_test)
test_loader = DataLoader(testset, batch_size=BATCH_SIZE, shuffle=False, num_workers=2)


# ==========================================
# 2. MODEL ARCHITECTURES (CIFAR-100 ADAPTED)
# ==========================================

class LeNet5CIFAR(nn.Module):
    """
    LeNet-5 adapted for 3-channel RGB 32x32 images with 100 target classes.
    """
    def __init__(self, num_classes=100):
        super(LeNet5CIFAR, self).__init__()
        # Input: 3x32x32 -> Conv1 -> 6x28x28
        self.conv1 = nn.Conv2d(3, 6, kernel_size=5)
        self.pool1 = nn.AvgPool2d(kernel_size=2, stride=2)  # -> 6x14x14
        self.conv2 = nn.Conv2d(6, 16, kernel_size=5)        # -> 16x10x10
        self.pool2 = nn.AvgPool2d(kernel_size=2, stride=2)  # -> 16x5x5

        self.fc1 = nn.Linear(16 * 5 * 5, 120)
        self.fc2 = nn.Linear(120, 84)
        self.fc3 = nn.Linear(84, num_classes)

    def forward(self, x):
        x = torch.tanh(self.conv1(x))
        x = self.pool1(x)
        x = torch.tanh(self.conv2(x))
        x = self.pool2(x)
        x = x.view(x.size(0), -1)
        x = torch.tanh(self.fc1(x))
        x = torch.tanh(self.fc2(x))
        x = self.fc3(x)
        return x


class AlexNetCIFAR(nn.Module):
    """
    AlexNet adapted for 32x32 input with 100 target classes.
    """
    def __init__(self, num_classes=100):
        super(AlexNetCIFAR, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 64, kernel_size=3, stride=1, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2),
            nn.Conv2d(64, 192, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2),
            nn.Conv2d(192, 384, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(384, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(256, 256, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )
        self.classifier = nn.Sequential(
            nn.Dropout(p=0.5),
            nn.Linear(256 * 4 * 4, 1024),
            nn.ReLU(inplace=True),
            nn.Dropout(p=0.5),
            nn.Linear(1024, 1024),
            nn.ReLU(inplace=True),
            nn.Linear(1024, num_classes),
        )

    def forward(self, x):
        x = self.features(x)
        x = torch.flatten(x, 1)
        x = self.classifier(x)
        return x


class VGG16CIFAR(nn.Module):
    """
    VGG16 adapted for 32x32 input with 100 target classes.
    """
    def __init__(self, num_classes=100):
        super(VGG16CIFAR, self).__init__()

        cfg = [64, 64, 'M', 128, 128, 'M', 256, 256, 256, 'M', 512, 512, 512, 'M', 512, 512, 512, 'M']
        self.features = self._make_layers(cfg)

        self.classifier = nn.Sequential(
            nn.Dropout(0.5),
            nn.Linear(512 * 1 * 1, 512),
            nn.ReLU(True),
            nn.Dropout(0.5),
            nn.Linear(512, 512),
            nn.ReLU(True),
            nn.Linear(512, num_classes),
        )

    def forward(self, x):
        x = self.features(x)
        x = torch.flatten(x, 1)
        x = self.classifier(x)
        return x

    def _make_layers(self, cfg):
        layers = []
        in_channels = 3
        for v in cfg:
            if v == 'M':
                layers += [nn.MaxPool2d(kernel_size=2, stride=2)]
            else:
                conv2d = nn.Conv2d(in_channels, v, kernel_size=3, padding=1)
                layers += [conv2d, nn.BatchNorm2d(v), nn.ReLU(inplace=True)]
                in_channels = v
        return nn.Sequential(*layers)


# ==========================================
# 3. UNIFIED TRAINING & EVALUATION ENGINE
# ==========================================

def train_and_evaluate(model_class, model_name):
    print(f"\n==========================================")
    print(f" Training Model: {model_name} (CIFAR-100)")
    print(f"==========================================")

    model = model_class(num_classes=NUM_CLASSES).to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE)

    start_time = time.time()

    for epoch in range(EPOCHS):
        model.train()
        running_loss = 0.0
        correct = 0
        total = 0

        for inputs, targets in train_loader:
            inputs, targets = inputs.to(device), targets.to(device)
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, targets)
            loss.backward()
            optimizer.step()

            running_loss += loss.item() * inputs.size(0)
            _, predicted = outputs.max(1)
            total += targets.size(0)
            correct += predicted.eq(targets).sum().item()

        train_acc = 100.0 * correct / total
        epoch_loss = running_loss / total

        # ==========================================
        # UPDATED EVALUATION PHASE (Top-1 & Top-5)
        # ==========================================
        model.eval()
        test_correct_top1 = 0
        test_correct_top5 = 0
        test_total = 0

        with torch.no_grad():
            for inputs, targets in test_loader:
                inputs, targets = inputs.to(device), targets.to(device)
                outputs = model(inputs)

                # 1. Top-1 predictions
                _, predicted = outputs.max(1)
                test_correct_top1 += predicted.eq(targets).sum().item()

                # 2. Top-5 predictions
                _, top5_preds = outputs.topk(5, dim=1, largest=True, sorted=True)
                # Reshape targets to (batch_size, 1) to match top5_preds matrix (batch_size, 5)
                test_correct_top5 += top5_preds.eq(targets.view(-1, 1)).sum().item()

                test_total += targets.size(0)

        test_acc_top1 = 100.0 * test_correct_top1 / test_total
        test_acc_top5 = 100.0 * test_correct_top5 / test_total

        print(f"Epoch [{epoch+1:02d}/{EPOCHS:02d}] | Loss: {epoch_loss:.4f} | Train Acc: {train_acc:.2f}% | Test Top-1: {test_acc_top1:.2f}% | Test Top-5: {test_acc_top5:.2f}%")

    elapsed_time = time.time() - start_time
    print(f"Finished {model_name} in {elapsed_time:.2f}s | Final Top-1: {test_acc_top1:.2f}% | Final Top-5: {test_acc_top5:.2f}%")
    return test_acc_top1, test_acc_top5, elapsed_time

# ==========================================
# 4. RUN COMPARISON
# ==========================================
if __name__ == '__main__':
    models_to_test = [
        (LeNet5CIFAR,"LeNet-5"),
        (AlexNetCIFAR,"AlexNet"),
        (VGG16CIFAR, "VGG16")
    ]

    results = {}
    for model_cls, name in models_to_test:
        acc, time_taken = train_and_evaluate(model_cls, name)
        results[name] = {"Accuracy": acc, "Time (s)": time_taken}

    print("\n" + "="*45)
    print("        FINAL COMPARISON SUMMARY (CIFAR-100)      ")
    print("="*45)
    print(f"{'Architecture':<15} | {'Test Accuracy':<15} | {'Training Time':<12}")
    print("-" * 45)
    for name, metrics in results.items():
        print(f"{name:<15} | {metrics['Accuracy']:.2f}%{'':<9} | {metrics['Time (s)']:.1f}s")
