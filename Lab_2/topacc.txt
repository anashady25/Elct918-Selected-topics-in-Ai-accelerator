# 1. Parse log data dynamically into structured dictionary
models_data = {}
current_model = None

model_header_pattern = re.compile(r"Training Model:\s*([^\(]+)")
epoch_pattern = re.compile(r"Epoch \[(\d+)/15\] \| Loss: ([\d\.]+) \| Train Acc: ([\d\.]+)% \| Test Top-1: ([\d\.]+)% \| Test Top-5: ([\d\.]+)%")

for line in log_text.strip().split("\n"):
    header_match = model_header_pattern.search(line)
    if header_match:
        current_model = header_match.group(1).strip()
        models_data[current_model] = {"epochs": [], "loss": [], "train_acc": [], "top1": [], "top5": []}
        continue

    epoch_match = epoch_pattern.search(line)
    if epoch_match and current_model:
        models_data[current_model]["epochs"].append(int(epoch_match.group(1)))
        models_data[current_model]["loss"].append(float(epoch_match.group(2)))
        models_data[current_model]["train_acc"].append(float(epoch_match.group(3)))
        models_data[current_model]["top1"].append(float(epoch_match.group(4)))
        models_data[current_model]["top5"].append(float(epoch_match.group(5)))

# 2. Configure 2x2 Plot Grid
fig, axes = plt.subplots(2, 2, figsize=(15, 11))
colors = {"LeNet-5": "#d62728", "AlexNet": "#1f77b4", "VGG16": "#2ca02c"}
markers = {"LeNet-5": "o", "AlexNet": "s", "VGG16": "^"}

# Subplot 1: Training Accuracy
ax_train = axes[0, 0]
for model_name, metrics in models_data.items():
    ax_train.plot(metrics["epochs"], metrics["train_acc"], label=model_name,
                  color=colors[model_name], marker=markers[model_name], linewidth=2)
ax_train.set_title("Training Accuracy Across Epochs", fontsize=12, fontweight="bold")
ax_train.set_xlabel("Epoch", fontsize=10)
ax_train.set_ylabel("Accuracy (%)", fontsize=10)
ax_train.set_xticks(range(1, 16))
ax_train.grid(True, linestyle=":", alpha=0.6)
ax_train.legend(fontsize=10)

# Subplot 2: Test Top-1 Accuracy
ax_top1 = axes[0, 1]
